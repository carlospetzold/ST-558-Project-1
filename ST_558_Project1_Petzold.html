response_to_tibble <- function(response) {
  httr::stop_for_status(response)

  response_text <- httr::content(
    response,
    as = "text",
    encoding = "UTF-8"
  )

  if (!jsonlite::validate(response_text)) {
    stop(
      "Census returned non-JSON content. Inspect response_text and check your API key.",
      call. = FALSE
    )
  }

  parsed_data <- jsonlite::fromJSON(response_text)

  column_names <- parsed_data[1, ]

  census_data <- tibble::as_tibble(
    parsed_data[-1, , drop = FALSE],
    .name_repair = "minimal"
  )

  names(census_data) <- column_names

  census_data[, !duplicated(names(census_data)), drop = FALSE]
}

Next, since the API returns many values as codes, we need the variable descriptions to know what they all are. We’ll write some functions below to extract the codes and labels from the metadata and standardize the codes before matching them:

# Return a vector where the names are codes and values are descriptions
get_labels <- function(variable_metadata) {
  labels <- unlist(
    variable_metadata$values$item,
    use.names = TRUE
  )

  if (length(labels) == 0L) {
    return(setNames(character(), character()))
  }

  labels
}

# Standardize numeric codes without losing zero
normalize_code <- function(x) {
  sub("^0+(?=[0-9])", "", as.character(x), perl = TRUE)
}

The JWAP and JWDP variables denote the arrival and departure times, but their values are time intervals and not actual clock times. We can use the midpoint of each interval as an approximate time. These functions convert the interval endpoints into seconds after midnight, calculate the midpoint, and store the result as a time of day.

# Convert a clock label, such as "4:10 a.m.", to seconds after midnight
clock_seconds <- function(x) {
  cleaned <- tolower(gsub(".", "", trimws(x), fixed = TRUE))

  pieces <- regmatches(
    cleaned,
    regexec("^([0-9]{1,2}):([0-9]{2})\\s*([ap]m)$", cleaned)
  )[[1]]

  if (length(pieces) != 4L) {
    stop("Unrecognized clock label: ", x)
  }

  hour <- as.numeric(pieces[2])
  minute <- as.numeric(pieces[3])

  if (hour < 1 || hour > 12 || minute > 59) {
    stop("Invalid clock label: ", x)
  }

  hour <- hour %% 12

  if (pieces[4] == "pm") {
    hour <- hour + 12
  }

  hour * 3600 + minute * 60
}

# Convert one interval description to its midpoint
interval_midpoint <- function(label) {
  if (is.na(label) ||
      grepl("N/A|not applicable|suppressed", label,
            ignore.case = TRUE)) {
    return(NA_real_)
  }

  endpoints <- strsplit(label, " to ", fixed = TRUE)[[1]]

  if (length(endpoints) != 2L) {
    stop("Unrecognized time interval: ", label)
  }

  start <- clock_seconds(endpoints[1])
  end <- clock_seconds(endpoints[2])

  # Allow an interval to cross midnight
  if (end < start) {
    end <- end + 24 * 3600
  }

  ((start + end) / 2) %% (24 * 3600)
}

# Decode the metadata once, then match the observed codes
parse_census_time <- function(x, variable_metadata) {
  labels <- get_labels(variable_metadata)

  midpoints <- vapply(
    unname(labels),
    interval_midpoint,
    numeric(1)
  )

  positions <- match(
    normalize_code(x),
    normalize_code(names(labels))
  )

  hms::hms(seconds = unname(midpoints[positions]))
}

Next, we need to give the remaining variables appropriate types: Numeric values to Numeric type, and categorical values to Factors. Some numeric values represent NA observations and are not actual measurements, and these can be identified using the metadata. Treating an NA code as an actual measurement would distort our mean and sd calculations, so it is critical that we assign these values as R NA values and not just leave them as is. That is specifically for the numeric values that we will be doing calculations on, and not the categorical variables since it would be fine to just leave the NA codes:

parse_census_numeric <- function(x, variable_metadata) {
  labels <- get_labels(variable_metadata)

  missing_codes <- names(labels)[
    grepl("N/A|not applicable|suppressed", labels,
          ignore.case = TRUE)
  ]

  x[normalize_code(x) %in% normalize_code(missing_codes)] <- NA_character_

  result <- suppressWarnings(as.numeric(x))

  # Recognize text null values, but catch unexpected text
  unexpected <- !is.na(x) &
    !tolower(trimws(x)) %in% c("", "null", "na", "nan") &
    is.na(result)

  if (any(unexpected)) {
    stop("Unexpected nonnumeric values in a numeric variable.")
  }

  result
}

parse_census_factor <- function(x, variable_metadata) {
  labels <- get_labels(variable_metadata)

  if (length(labels) == 0L) {
    stop("No categorical labels found in the metadata.")
  }

  labels <- labels[order(as.numeric(names(labels)))]

  unknown <- !is.na(x) &
    !normalize_code(x) %in% normalize_code(names(labels))

  if (any(unknown)) {
    stop("A categorical code was not found in the metadata.")
  }

  factor(
    normalize_code(x),
    levels = normalize_code(names(labels)),
    labels = unname(labels)
  )
}

Now, we’ll combine all of our functions above to create the main function query_census. The user can choose a survey year, numeric variables, categorical variables, and geography. Before making a request, the function checks that the inputs meet the proper requirements, specifically: requires PWGTP, at least one other numeric variable, and at least one categorical variable.

The default request uses 2024 data for NC, returning age, sex, and weight. Users can also elect a region/division, or request all valid values within a geography type. Geo codes are checked against the metadata, with PR excluded. Lastly, the function converts the returned columns and adds the “census” class, which allows our custom summary and plotting methods to recognize the data.

query_census <- function(year = 2024, numeric_vars = c("AGEP", "PWGTP"), categorical_vars = "SEX", geography = "State", geography_values = NULL) {

  # Validate the year
  if (!is.numeric(year) ||
      length(year) != 1L ||
      anyNA(year) ||
      !year %in% 2021:2024) {
    stop("year must be one of 2021, 2022, 2023, or 2024.")
  }

  allowed_numeric <- c(
    "AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP"
  )

  allowed_categorical <- c("FER", "HHL", "SCH", "SCHL", "SEX")

  # Validate numeric variable selections
  if (!is.character(numeric_vars) ||
      anyNA(numeric_vars) ||
      !all(numeric_vars %in% allowed_numeric)) {
    stop("numeric_vars contains an unsupported variable.")
  }

  if (!"PWGTP" %in% numeric_vars ||
      length(setdiff(numeric_vars, "PWGTP")) == 0L) {
    stop("Include PWGTP and at least one other numeric variable.")
  }

  # Validate categorical variable selections
  if (!is.character(categorical_vars) ||
      length(categorical_vars) == 0L ||
      anyNA(categorical_vars) ||
      !all(categorical_vars %in% allowed_categorical)) {
    stop("Select at least one supported categorical variable.")
  }

  numeric_vars <- unique(numeric_vars)
  categorical_vars <- unique(categorical_vars)

  # Validate the geography type
  if (!is.character(geography) ||
      length(geography) != 1L ||
      anyNA(geography) ||
      !geography %in% c("Region", "Division", "State")) {
    stop("geography must be exactly Region, Division, or State.")
  }

  # Load the correct year's metadata
  metadata <- readRDS("data/census_var_list.rds")[[as.character(year)]]

  if (is.null(metadata)) {
    stop("Metadata for the requested year was not found.")
  }

  geography_variable <- switch(
    geography,
    Region = "REGION",
    Division = "DIVISION",
    State = if (year <= 2022) "ST" else "STATE"
  )

  geography_labels <- get_labels(metadata[[geography_variable]])

  # Puerto Rico is outside the scope of this project
  geography_labels <- geography_labels[
    !grepl("Puerto Rico", geography_labels, ignore.case = TRUE)
  ]

  valid_codes <- names(geography_labels)

  # Defaults: North Carolina, South Atlantic, or South
  if (is.null(geography_values)) {
    geography_values <- switch(
      geography,
      State = "37",
      Division = "5",
      Region = "3"
    )
  }

  if (!(is.character(geography_values) ||
        is.numeric(geography_values)) ||
      length(geography_values) == 0L ||
      anyNA(geography_values)) {
    stop("Supply valid geography codes or 'all'.")
  }

  # 'all' is a selection within a geography type
  if (length(geography_values) == 1L &&
      tolower(as.character(geography_values)) == "all") {

    selected_codes <- valid_codes

  } else {

    positions <- match(
      normalize_code(geography_values),
      normalize_code(valid_codes)
    )

    if (anyNA(positions)) {
      stop("Invalid geography code for this year and geography type.")
    }

    selected_codes <- unique(valid_codes[positions])
  }

  api_key <- trimws(Sys.getenv("CENSUS_API_KEY"))

  if (!nzchar(api_key)) {
    stop("Set CENSUS_API_KEY in your R session before querying.")
  }

  response <- httr::GET(
    paste0("https://api.census.gov/data/", year, "/acs/acs1/pums"),
    query = list(
      get = paste(c(numeric_vars, categorical_vars), collapse = ","),
      `for` = paste0(
        tolower(geography), ":",
        paste(selected_codes, collapse = ",")
      ),
      key = api_key
    )
  )

  result <- response_to_tibble(response)

  required_columns <- c(
    numeric_vars, categorical_vars, tolower(geography)
  )

  if (!all(required_columns %in% names(result))) {
    stop("The API response is missing requested columns.")
  }

  # Convert ordinary numeric variables and time variables separately
  for (variable in numeric_vars) {
    if (variable %in% c("JWAP", "JWDP")) {
      result[[variable]] <- parse_census_time(
        result[[variable]], metadata[[variable]]
      )
    } else {
      result[[variable]] <- parse_census_numeric(
        result[[variable]], metadata[[variable]]
      )
    }
  }

  for (variable in categorical_vars) {
    result[[variable]] <- parse_census_factor(
      result[[variable]], metadata[[variable]]
    )
  }

  # Give geography codes readable labels
  geography_column <- tolower(geography)

  result[[geography_column]] <- parse_census_factor(
    result[[geography_column]],
    metadata[[geography_variable]]
  )

  class(result) <- c("census", class(result))

  result
}

Our single-year function handles one survey year at a time. To compare years, we will write a wrapper that calls it once for each requested year and combines the results. Reusing the existing function keeps the validation and conversion steps consistent. We will also add a year column and store it as a factor because it identifies survey groups for our summaries and plots:

query_census_years <- function(years = 2021:2024, numeric_vars = c("AGEP", "PWGTP"), categorical_vars = "SEX", geography = "State", geography_values = NULL) {

  if (!is.numeric(years) ||
      length(years) == 0L ||
      anyNA(years) ||
      !all(years %in% 2021:2024)) {
    stop("years must contain values from 2021 through 2024.")
  }

  years <- unique(years)

  yearly_data <- lapply(years, function(current_year) {

    result <- query_census(
      year = current_year,
      numeric_vars = numeric_vars,
      categorical_vars = categorical_vars,
      geography = geography,
      geography_values = geography_values
    )

    # Bind ordinary tibbles, then restore the custom class
    class(result) <- setdiff(class(result), "census")
    result$year <- as.character(current_year)

    result
  })

  result <- dplyr::bind_rows(yearly_data)
  result$year <- factor(result$year, levels = as.character(sort(years)))

  class(result) <- c("census", class(result))

  result
}

Part 2: Writing a Generic Function for Summarizing

Now we will work on creating some generic functions that we can use to explore our census data, specifically adding methods to the existing summary() and plot() generic functions. We will start with 2 smaller helper functions to identify eligible columns and validate weights. Time variables should not be included in the numeric summaries, and weights must be numeric, non-negative, and finite so that weighted calculations are meaningful.

Each row represents a sampled person, and PWGTP indicates how many people that record represents in the population estimate. Giving every row equal importance would ignore these weights. Our summary and plotting methods will therefore use PWGTP to account for each record’s contribution without physically repeating rows:

is_measurement <- function(x) {
  is.numeric(x) && !inherits(x, c("difftime", "POSIXt", "Date"))
}

check_weights <- function(x) {
  if (!"PWGTP" %in% names(x) ||
      !is.numeric(x$PWGTP) ||
      anyNA(x$PWGTP) ||
      any(!is.finite(x$PWGTP)) ||
      any(x$PWGTP < 0)) {
    stop("PWGTP must contain finite, nonnegative numeric weights.")
  }
}

Each row has a person’s weight, PWGTP, indicating how much it contributes to the population estimates. Treating every row equally would ignore these differences, so our summary method calculates weighted means and standard deviations using the given formulas, and categorical counts by adding the weights within each category.

By default, the method summarizes all eligible numeric and categorical columns, excluding PWGTP and time variables from the numeric calculations. For each numeric variable, observations with missing values are excluded along with their weights. The results are returned as a named list, separating numeric summaries from categorical counts.

summary.census <- function(
    object,
    numeric_vars = NULL,
    categorical_vars = NULL,
    ...
) {

  check_weights(object)

  available_numeric <- setdiff(
    names(object)[vapply(object, is_measurement, logical(1))],
    c("PWGTP", "year")
  )

  available_categorical <- names(object)[
    vapply(
      object,
      function(x) is.factor(x) || is.character(x),
      logical(1)
    )
  ]

  if (is.null(numeric_vars)) {
    numeric_vars <- available_numeric
  }

  if (is.null(categorical_vars)) {
    categorical_vars <- available_categorical
  }

  if (!is.character(numeric_vars) ||
      anyNA(numeric_vars) ||
      !all(numeric_vars %in% available_numeric)) {
    stop("Select eligible numeric variables; exclude PWGTP and times.")
  }

  if (!is.character(categorical_vars) ||
      anyNA(categorical_vars) ||
      !all(categorical_vars %in% available_categorical)) {
    stop("Select categorical columns present in the data.")
  }

  numeric_vars <- unique(numeric_vars)
  categorical_vars <- unique(categorical_vars)

  numeric_results <- lapply(numeric_vars, function(variable) {
    values <- object[[variable]]
    weights <- object$PWGTP

    # Use the same observations in numerator and denominator
    keep <- is.finite(values) & weights > 0

    values <- values[keep]
    weights <- weights[keep]

    if (length(values) == 0L) {
      return(c(mean = NA_real_, sd = NA_real_))
    }

    weighted_mean <- sum(values * weights) / sum(weights)

    weighted_variance <- (
      sum(values^2 * weights) / sum(weights)
    ) - weighted_mean^2

    c(
      mean = weighted_mean,
      sd = sqrt(max(0, weighted_variance))
    )
  })

  names(numeric_results) <- numeric_vars

  categorical_results <- lapply(categorical_vars, function(variable) {
    values <- object[[variable]]

    if (!is.factor(values)) {
      values <- factor(values)
    }

    counts <- vapply(levels(values), function(level) {
      sum(object$PWGTP[!is.na(values) & values == level])
    }, numeric(1))

    # Report actual missing observations separately
    if (anyNA(values)) {
      counts <- c(
        counts,
        "(Missing)" = sum(object$PWGTP[is.na(values)])
      )
    }

    counts
  })

  names(categorical_results) <- categorical_vars

  list(
    numeric = numeric_results,
    categorical = categorical_results
  )
}

To visually compare, we will create a plotting method for objects with the “census” class. We must select one categorical variable and one numeric measurement. The method produces a boxplot for each category and includes PWGTP in the weight aesthetic so that the boxplot calculations account for the person weights. Observations missing either selected variable, or having zero weight, are excluded from the plot.

plot.census <- function(x, cat_var, num_var, ...) {

  check_weights(x)

  if (missing(cat_var) || missing(num_var)) {
    stop("Specify one categorical variable and one numeric variable.")
  }

  if (!is.character(cat_var) ||
      length(cat_var) != 1L ||
      anyNA(cat_var) ||
      !cat_var %in% names(x)) {
    stop("cat_var must name one column in the data.")
  }

  if (!is.character(num_var) ||
      length(num_var) != 1L ||
      anyNA(num_var) ||
      !num_var %in% names(x)) {
    stop("num_var must name one column in the data.")
  }

  if (!(is.factor(x[[cat_var]]) || is.character(x[[cat_var]]))) {
    stop("cat_var must be categorical.")
  }

  if (!is_measurement(x[[num_var]]) ||
      num_var %in% c("PWGTP", "year")) {
    stop("num_var must be a numeric measurement, excluding weights and times.")
  }

  keep <- !is.na(x[[cat_var]]) &
    is.finite(x[[num_var]]) &
    x$PWGTP > 0

  plot_data <- x[keep, ]

  if (nrow(plot_data) == 0L) {
    stop("No complete observations with positive weights to plot.")
  }

  class(plot_data) <- setdiff(class(plot_data), "census")

  ggplot2::ggplot(
    plot_data,
    ggplot2::aes(
      x = .data[[cat_var]],
      y = .data[[num_var]],
      weight = PWGTP
    )
  ) +
    ggplot2::geom_boxplot(...) +
    ggplot2::labs(
      x = cat_var,
      y = num_var,
      title = paste(num_var, "by", cat_var),
      subtitle = "Weighted using PWGTP"
    ) +
    ggplot2::theme_minimal()
}

Investigation

Now let’s investigate something that we are interested in within our data. After the most intense portion of the COVID pandemic (2020) ended, many people continued to work from home, but others were asked to return to the office after the mandates were lifted. I would be interested in seeing if the lingering effects of the pandemic had effects on commute times for people, and if these changed over the 2021-2024 period as more people went back into the office. It could also be interesting to see how these commute times differed by sex, which could indicate if the jobs that are primarily held by either men or women had any differences in how the pandemic affected them. Let’s also use just the information for North Carolina to see the trends specifically for our state, and to keep a reasonable amount of values in our data.

But something else to be aware of is that fact that, if someone was working from home during the pandemic, then eventually was asked to return to the office, they may have been an NA value until the point when the returned to the office, and then finally reported a commute time when they did. This is likely a very common occurrence, so something else that could be considered in our analysis is what proportion of the respondents were NA across the 4 years, which we look at below (not using our functions):
